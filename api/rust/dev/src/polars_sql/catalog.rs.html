38/// A statement changing a catalog table, for the caller to perform (see
39/// [`SQLContext::catalog_ddl`](crate::SQLContext::catalog_ddl)).
40#[derive(Clone, Debug, PartialEq, Eq)]
41pub enum CatalogDdl {
42    /// `DROP TABLE [IF EXISTS] <name> [PURGE]`.
43    DropTable {
44        /// The table name as the statement wrote it.
45        name: String,
46        /// The catalog table it resolves to.
47        table: CatalogTable,
48        /// `IF EXISTS`.
49        if_exists: bool,
50        /// `PURGE`: also delete the data files.
51        purge: bool,
52    },
53    /// `CREATE TABLE [IF NOT EXISTS] <name> (<column>, ...)`, or `... <name> AS <query>`.
54    CreateTable {
55        /// The table name as the statement wrote it.
56        name: String,
57        /// The catalog table to create.
58        table: CatalogTable,
59        /// `IF NOT EXISTS`.
60        if_not_exists: bool,
61        /// The columns; empty with a query.
62        schema: Schema,
63        /// The query that fills the table, as SQL.
64        query: Option<String>,
65        /// The columns declared `NOT NULL`.
66        required: Vec<String>,
67        /// The table location, from `LOCATION` or the `location` option.
68        location: Option<String>,
69        /// The other `WITH` options, as table properties.
70        properties: Vec<(String, String)>,
71        /// The partition fields, in the order written; empty for an unpartitioned table.
72        partition_by: Vec<CatalogPartitionField>,
73    },
74    /// `CREATE SCHEMA [IF NOT EXISTS] <name> [WITH (...)]`.
75    CreateSchema {
76        /// The schema name as the statement wrote it.
77        name: String,
78        /// The catalog namespace to create.
79        namespace: CatalogNamespace,
80        /// `IF NOT EXISTS`.
81        if_not_exists: bool,
82        /// The `WITH` options, as namespace properties.
83        properties: Vec<(String, String)>,
84    },
85    /// `DROP SCHEMA [IF EXISTS] <name> [RESTRICT]`.
86    DropSchema {
87        /// The schema name as the statement wrote it.
88        name: String,
89        /// The catalog namespace to drop.
90        namespace: CatalogNamespace,
91        /// `IF EXISTS`.
92        if_exists: bool,
93    },
94}

131/// The catalogs attached to a context.
132#[derive(Clone, Debug, Default)]
133pub(crate) struct CatalogRegistry {
134    catalogs: Vec<AttachedCatalog>,
135    /// The catalog of table names that don't start with an attached catalog.
136    default: Option<String>,
137}
138
139impl CatalogRegistry {
140    pub(crate) fn is_empty(&self) -> bool {
141        self.catalogs.is_empty()
142    }
143
144    /// Attach a catalog; the same name replaces it, with the new `default_namespace`, and keeps
145    /// its default status (`default` only ever sets the default). A name that differs from an
146    /// attached one only in case is an error, so matching names ignoring case is never ambiguous.
147    pub(crate) fn register(
148        &mut self,
149        name: &str,
150        default_namespace: Option<Vec<String>>,
151        default: bool,
152    ) -> PolarsResult<()> {
153        polars_ensure!(!name.is_empty(), InvalidOperation: "catalog name must not be empty");
154        polars_ensure!(
155            default_namespace.as_ref().is_none_or(|ns| !ns.is_empty() && ns.iter().all(|p| !p.is_empty())),
156            InvalidOperation: "the default namespace of catalog '{}' must have non-empty parts", name
157        );
158        if let Some(other) = self
159            .catalogs
160            .iter()
161            .find(|c| c.name != name && c.name.eq_ignore_ascii_case(name))
162        {
163            polars_bail!(
164                InvalidOperation: "catalog '{}' differs from attached catalog '{}' only in case; use that name, or unregister it first",
165                name, other.name
166            );
167        }
168        let keep_default = self.default.as_deref() == Some(name);
169        self.unregister(name);
170        self.catalogs.push(AttachedCatalog {
171            name: name.to_string(),
172            default_namespace,
173        });
174        if default || keep_default {
175            self.default = Some(name.to_string());
176        }
177        Ok(())
178    }
179
180    /// The name of the default catalog.
181    pub(crate) fn default_catalog(&self) -> Option<&str> {
182        self.default.as_deref()
183    }
184
185    /// Make an attached catalog (exact name, else ignoring case) the default.
186    pub(crate) fn set_default(&mut self, name: &str) -> PolarsResult<()> {
187        let Some(name) = self.catalog_name(name).map(str::to_string) else {
188            polars_bail!(SQLInterface: "catalog '{}' is not attached", name);
189        };
190        self.default = Some(name);
191        Ok(())
192    }
193
194    /// Detach a catalog (exact name, else ignoring case), returning the name it had.
195    pub(crate) fn unregister(&mut self, name: &str) -> Option<String> {
196        let idx = self.position(name)?;
197        let removed = self.catalogs.remove(idx).name;
198        if self.default.as_deref() == Some(removed.as_str()) {
199            self.default = None;
200        }
201        Some(removed)
202    }
203
204    fn position(&self, name: &str) -> Option<usize> {
205        self.catalogs
206            .iter()
207            .position(|c| c.name == name)
208            .or_else(|| {
209                self.catalogs
210                    .iter()
211                    .position(|c| c.name.eq_ignore_ascii_case(name))
212            })
213    }
214
215    /// The attached catalog called `name` (exact, else ignoring case).
216    pub(crate) fn catalog_name(&self, name: &str) -> Option<&str> {
217        self.position(name).map(|i| self.catalogs[i].name.as_str())
218    }
219
220    /// The attached catalog a table name explicitly starts with.
221    pub(crate) fn named_catalog(&self, parts: &[String]) -> Option<&str> {
222        match parts {
223            [first, _, ..] => self.catalog_name(first),
224            _ => None,
225        }
226    }
227
228    /// The catalog table a table name refers to: a leading part naming an attached catalog
229    /// selects it, else the default catalog; a name without namespace parts gets the catalog's
230    /// default namespace. `None` if no catalog applies.
231    pub(crate) fn resolve(&self, parts: &[String]) -> Option<CatalogTable> {
232        let (catalog, rest) = self.select(parts)?;
233        let identifier = match rest {
234            [table] => {
235                let mut identifier = catalog.default_namespace.clone()?;
236                identifier.push(table.clone());
237                identifier
238            },
239            _ => rest.to_vec(),
240        };
241        Some(CatalogTable {
242            catalog: catalog.name.clone(),
243            identifier,
244        })
245    }
246
247    /// The catalog namespace a schema name refers to: a leading part naming an attached catalog
248    /// selects it, else the default catalog; the other parts are the namespace (never relative
249    /// to a default namespace). `None` if no catalog applies.
250    pub(crate) fn resolve_namespace(&self, parts: &[String]) -> Option<CatalogNamespace> {
251        let (catalog, rest) = self.select(parts)?;
252        Some(CatalogNamespace {
253            catalog: catalog.name.clone(),
254            namespace: rest.to_vec(),
255        })
256    }
257
258    /// The catalog a name selects, and the name's other parts.
259    fn select<'a>(&self, parts: &'a [String]) -> Option<(&AttachedCatalog, &'a [String])> {
260        let named = match parts {
261            [first, rest @ ..] if !rest.is_empty() => self.position(first).map(|idx| (idx, rest)),
262            _ => None,
263        };
264        let (idx, rest) = match named {
265            Some(named) => named,
266            None => (self.position(self.default.as_deref()?)?, parts),
267        };
268        Some((&self.catalogs[idx], rest))
269    }
270
271    /// Why a table name that nothing resolved is not a catalog table, if a catalog could have
272    /// applied.
273    pub(crate) fn unresolved_hint(&self, parts: &[String]) -> Option<String> {
274        if self.is_empty() || self.resolve(parts).is_some() {
275            return None;
276        }
277        let catalog = match self.named_catalog(parts) {
278            Some(named) => named,
279            None => match &self.default {
280                Some(default) => default.as_str(),
281                None => return Some("; no default catalog is set".to_string()),
282            },
283        };
284        Some(format!("; catalog '{catalog}' has no default namespace"))
285    }
286}

299/// The names a statement wrote for its catalog tables, by the position of each rewritten name

309/// Rewrite the names of catalog tables in `stmt` to their full names, `lineitem` to
310/// `onelake.tpch.lineitem`, so the rest of the translation knows them by that name. Names of
311/// CTEs in scope and of local tables (`is_local`) are left alone. Returns each rewritten name
312/// as written with its full name, and where each name was written.
313pub(crate) fn rewrite_catalog_names(
314    stmt: &mut Statement,
315    resolve: impl Fn(&[String]) -> Option<Vec<String>>,
316    is_local: impl Fn(&str) -> bool,
317) -> PolarsResult<(Vec<RewrittenName>, CatalogSpellings)> {
318    let mut registered = RegisteredAliases(Vec::new());
319    let _ = dyn_visit::visit(&*stmt, &mut registered);
320    let mut rewriter = Rewriter {
321        resolve,
322        is_local,
323        registered_aliases: registered.0,
324        cte_scopes: Vec::new(),
325        with_stack: Vec::new(),
326        rewritten: Vec::new(),
327        spellings: CatalogSpellings::default(),
328        desugared: 0,
329    };
330    if let ControlFlow::Break(err) = dyn_visit::visit_mut(stmt, &mut rewriter) {
331        return Err(err);
332    }
333    Ok((rewriter.rewritten, rewriter.spellings))
334}

336/// The aliases of the statement's derived tables, table functions and `UNNEST`s, which are
337/// registered as tables while the statement runs (#30070), wherever they are written.
338struct RegisteredAliases(Vec<String>);
339
340impl Visitor for RegisteredAliases {
341    type Break = ();
342
343    fn pre_visit_table_factor(&mut self, factor: &TableFactor) -> ControlFlow<()> {
344        let alias = match factor {
345            TableFactor::Table {
346                args: Some(_),
347                alias,
348                ..
349            }
350            | TableFactor::Derived { alias, .. }
351            | TableFactor::UNNEST { alias, .. } => alias.as_ref(),
352            _ => None,
353        };
354        self.0.extend(alias.map(|alias| alias.name.value.clone()));
355        ControlFlow::Continue(())
356    }
357}

359struct Rewriter<R, L> {
360    resolve: R,
361    is_local: L,
362    /// Aliases that tables of the same name would be looked up as (see [`RegisteredAliases`]).
363    registered_aliases: Vec<String>,
364    /// The CTE names visible in each enclosing query.
365    cte_scopes: Vec<Vec<String>>,
366    /// The `WITH` clauses of the queries being visited, taken out while their bodies are visited.
367    with_stack: Vec<Option<With>>,
368    rewritten: Vec<RewrittenName>,
369    spellings: CatalogSpellings,
370    /// The number of `TABLE x` desugared, which numbers their (made-up) name positions.
371    desugared: u64,
372}
373
374impl<R, L> Rewriter<R, L>
375where
376    R: Fn(&[String]) -> Option<Vec<String>>,
377    L: Fn(&str) -> bool,
378{
379    fn is_cte(&self, name: &str) -> bool {
380        self.cte_scopes
381            .iter()
382            .flatten()
383            .any(|cte| cte.eq_ignore_ascii_case(name))
384    }
385
386    /// The full name of the catalog table that `parts` refers to: not a CTE or a local table.
387    fn resolve_name(&self, parts: &[String]) -> Option<Vec<String>> {
388        let written = parts.join(".");
389        if self.is_cte(&written) || (self.is_local)(&written) {
390            return None;
391        }
392        (self.resolve)(parts)
393    }
394
395    /// Record `parts` as referring to the catalog table `full`. Tables are looked up by their
396    /// dotted name, so it must be the catalog table's alone.
397    fn record(&mut self, parts: &[String], full: &[String]) -> PolarsResult<()> {
398        let key = full.join(".");
399        let is_alias = self
400            .registered_aliases
401            .iter()
402            .any(|alias| alias.eq_ignore_ascii_case(&key));
403        if is_alias || self.is_cte(&key) || (self.is_local)(&key) {
404            polars_bail!(
405                SQLInterface: "table '{}' refers to catalog table '{}', which is also the name of a registered table, CTE or alias; rename one of them",
406                display_name(parts), display_name(full)
407            );
408        }
409        if let Some((_, other)) = self
410            .rewritten
411            .iter()
412            .find(|(_, other)| other.join(".") == key && other != full)
413        {
414            polars_bail!(
415                SQLInterface: "catalog tables {} and {} have the same dotted name, which SQL cannot tell apart",
416                display_name(other), display_name(full)
417            );
418        }
419        if !self.rewritten.iter().any(|(written, _)| written == parts) {
420            self.rewritten.push((parts.to_vec(), full.to_vec()));
421        }
422        Ok(())
423    }
424
425    /// Replace `TABLE x` of a catalog table by `SELECT * FROM x`, whose table name the visit
426    /// then rewrites (sqlparser's `TABLE` holds at most two name parts).
427    fn desugar_table(&mut self, body: &mut SetExpr) {
428        match body {
429            SetExpr::SetOperation { left, right, .. } => {
430                self.desugar_table(left);
431                self.desugar_table(right);
432            },
433            SetExpr::Table(table) => {
434                let parts: Vec<String> = table
435                    .schema_name
436                    .iter()
437                    .chain(&table.table_name)
438                    .cloned()
439                    .collect();
440                if !parts.is_empty() && self.resolve_name(&parts).is_some() {
441                    // Line 0 is never a position in the query text.
442                    self.desugared += 1;
443                    let at = Location::new(0, self.desugared);
444                    *body = select_star_from(&parts, Span::new(at, at));
445                }
446            },
447            _ => {},
448        }
449    }
450}

452impl<R, L> VisitorMut for Rewriter<R, L>
453where
454    R: Fn(&[String]) -> Option<Vec<String>>,
455    L: Fn(&str) -> bool,
456{
457    type Break = PolarsError;
458
459    fn pre_visit_query(&mut self, query: &mut Query) -> ControlFlow<PolarsError> {
460        // A non-recursive CTE is visible after its definition, as `register_ctes` registers them.
461        let mut with = query.with.take();
462        let mut names = Vec::new();
463        if let Some(with) = with.as_mut() {
464            let all: Vec<String> = with
465                .cte_tables
466                .iter()
467                .map(|cte| cte.alias.name.value.clone())
468                .collect();
469            let recursive = with.recursive;
470            for cte in &mut with.cte_tables {
471                self.cte_scopes.push(if recursive {
472                    all.clone()
473                } else {
474                    names.clone()
475                });
476                let flow = dyn_visit::visit_mut(&mut *cte.query, self);
477                self.cte_scopes.pop();
478                flow?;
479                names.push(cte.alias.name.value.clone());
480            }
481        }
482        self.cte_scopes.push(names);
483        self.with_stack.push(with);
484        self.desugar_table(&mut query.body);
485        ControlFlow::Continue(())
486    }
487
488    fn post_visit_query(&mut self, query: &mut Query) -> ControlFlow<PolarsError> {
489        self.cte_scopes.pop();
490        query.with = self.with_stack.pop().flatten();
491        ControlFlow::Continue(())
492    }
493
494    fn pre_visit_table_factor(&mut self, factor: &mut TableFactor) -> ControlFlow<PolarsError> {
495        // A name with arguments is a table function.
496        let TableFactor::Table {
497            name, args: None, ..
498        } = factor
499        else {
500            return ControlFlow::Continue(());
501        };
502        let Some(parts) = name_parts(name) else {
503            return ControlFlow::Continue(());
504        };
505        if let Some(full) = self.resolve_name(&parts) {
506            if let Err(err) = self.record(&parts, &full) {
507                return ControlFlow::Break(err);
508            }
509            let span = name.0[0]
510                .as_ident()
511                .map_or_else(Span::empty, |ident| ident.span);
512            if let Some(position) = name_position(name) {
513                self.spellings.insert(position, parts);
514            }
515            *name = object_name(&full, span);
516        }
517        ControlFlow::Continue(())
518    }
519}

521/// A table name whose parts all have `span`, so the name keeps the position it was written at.
522fn object_name(parts: &[String], span: Span) -> ObjectName {
523    let ident = |part: &String| Ident {
524        value: part.clone(),
525        quote_style: None,
526        span,
527    };
528    ObjectName(
529        parts
530            .iter()
531            .map(|part| ObjectNamePart::Identifier(ident(part)))
532            .collect(),
533    )
534}

536/// `SELECT * FROM <parts>`, with the name at `span`.
537fn select_star_from(parts: &[String], span: Span) -> SetExpr {
538    let mut stmts = Parser::parse_sql(&GenericDialect, "SELECT * FROM t").unwrap();
539    let Statement::Query(query) = stmts.remove(0) else {
540        unreachable!()
541    };
542    let mut body = *query.body;
543    if let SetExpr::Select(select) = &mut body
544        && let TableFactor::Table { name, .. } = &mut select.from[0].relation
545    {
546        *name = object_name(parts, span);
547    }
548    body
549}

551/// The schema of a catalog table from its column definitions, and its `NOT NULL` columns.
552pub(crate) fn create_table_columns(columns: &[ColumnDef]) -> PolarsResult<(Schema, Vec<String>)> {
553    let mut schema = Schema::with_capacity(columns.len());
554    let mut required = Vec::new();
555    for col in columns {
556        let name = col.name.value.as_str();
557        let mut not_null = None;
558        for option in &col.options {
559            let is_not_null = match option.option {
560                ColumnOption::NotNull => true,
561                ColumnOption::Null => false,
562                _ => polars_bail!(
563                    SQLInterface: "only NOT NULL constraints are supported for catalog tables; found `{}` on column '{}'",
564                    option, name
565                ),
566            };
567            polars_ensure!(
568                not_null.is_none_or(|n| n == is_not_null),
569                SQLInterface: "conflicting NULL/NOT NULL declarations for column '{}'", name
570            );
571            not_null = Some(is_not_null);
572        }
573        let dtype = map_sql_dtype_to_polars(&col.data_type)?;
574        polars_ensure!(
575            schema.insert(name.into(), dtype).is_none(),
576            Duplicate: "column '{}' is defined more than once", name
577        );
578        if not_null == Some(true) {
579            required.push(name.to_string());
580        }
581    }
582    Ok((schema, required))
583}

585/// Iceberg's reserved table properties, other than `format-version`, which a create may set.
586const RESERVED_TABLE_PROPERTIES: [&str; 8] = [
587    "uuid",
588    "snapshot-count",
589    "current-snapshot-id",
590    "current-snapshot-summary",
591    "current-snapshot-timestamp-ms",
592    "current-schema",
593    "default-partition-spec",
594    "default-sort-order",
595];

597/// Table properties, in the order written.
598type TableProperties = Vec<(String, String)>;
599
600/// The location and properties of a catalog table, from `LOCATION` and `WITH (...)`.
601pub(crate) fn create_table_options(
602    options: &CreateTableOptions,
603    location_clause: Option<&str>,
604) -> PolarsResult<(Option<String>, TableProperties)> {
605    let options = match options {
606        CreateTableOptions::None => &[][..],
607        CreateTableOptions::With(options) => options.as_slice(),
608        CreateTableOptions::TableProperties(_) => {
609            polars_bail!(SQLInterface: "`TBLPROPERTIES` is not supported; use `WITH (...)` instead")
610        },
611        _ => {
612            polars_bail!(SQLInterface: "table `OPTIONS` clauses are not supported; use `WITH (...)` instead")
613        },
614    };
615    let mut location = location_clause.map(str::to_string);
616    let mut properties = TableProperties::new();
617    for option in options {
618        let (key, value) = option_key_value(option, "table", &["location", "format-version"])?;
619        polars_ensure!(
620            !RESERVED_TABLE_PROPERTIES.contains(&key),
621            SQLInterface: "table option '{}' is reserved by Iceberg", key
622        );
623        if key == "location" {
624            polars_ensure!(
625                location_clause.is_none(),
626                SQLInterface: "`LOCATION` and table option 'location' cannot be combined"
627            );
628            polars_ensure!(location.is_none(), SQLInterface: "duplicate table option 'location'");
629            location = Some(value);
630        } else {
631            polars_ensure!(
632                properties.iter().all(|(k, _)| k != key),
633                SQLInterface: "duplicate table option '{}'", key
634            );
635            properties.push((key.to_string(), value));
636        }
637    }
638    if let Some(location) = &location {
639        polars_ensure!(
640            is_absolute_location(location),
641            SQLInterface: "location must be an absolute path or a URI; found '{}'", location
642        );
643    }
644    Ok((location, properties))
645}

647/// The partition fields of a catalog table, from `PARTITION BY (...)` or `PARTITIONED BY (...)`.
648pub(crate) fn create_table_partitioning(
649    partition_by: Option<&Expr>,
650    hive_distribution: &HiveDistributionStyle,
651) -> PolarsResult<Vec<CatalogPartitionField>> {
652    // Each field with its SQL, for messages.
653    let fields: Vec<(CatalogPartitionField, String)> = match (partition_by, hive_distribution) {
654        (Some(_), HiveDistributionStyle::PARTITIONED { .. }) => {
655            polars_bail!(SQLInterface: "`PARTITION BY` and `PARTITIONED BY` cannot be combined")
656        },
657        (Some(expr), _) => {
658            let exprs = match unparenthesized(expr) {
659                Expr::Tuple(exprs) => exprs.as_slice(),
660                expr => std::slice::from_ref(expr),
661            };
662            exprs
663                .iter()
664                .map(|expr| Ok((partition_field(expr)?, expr.to_string())))
665                .collect::<PolarsResult<_>>()?
666        },
667        (None, HiveDistributionStyle::PARTITIONED { columns }) => columns
668            .iter()
669            .map(|col| {
670                // Hive's `PARTITIONED BY (p INT)` adds a column.
671                polars_ensure!(
672                    col.data_type == SQLDataType::Unspecified && col.options.is_empty(),
673                    SQLInterface: "`PARTITIONED BY` takes column names only; found `{}`; types and constraints belong in the column list",
674                    col
675                );
676                let field = CatalogPartitionField {
677                    column: col.name.value.clone(),
678                    transform: "identity".to_string(),
679                };
680                Ok((field, col.name.value.clone()))
681            })
682            .collect::<PolarsResult<_>>()?,
683        (None, _) => Vec::new(),
684    };
685    // Iceberg allows one time transform per column.
686    let is_time = |f: &CatalogPartitionField| {
687        matches!(f.transform.as_str(), "year" | "month" | "day" | "hour")
688    };
689    for (i, (field, sql)) in fields.iter().enumerate() {
690        for (earlier, earlier_sql) in &fields[..i] {
691            polars_ensure!(earlier != field, SQLInterface: "duplicate partition field `{}`", sql);
692            polars_ensure!(
693                earlier.column != field.column || !is_time(earlier) || !is_time(field),
694                SQLInterface: "partition fields `{}` and `{}` both partition column '{}' by time; use one of them",
695                earlier_sql, sql, field.column
696            );
697        }
698    }
699    Ok(fields.into_iter().map(|(field, _)| field).collect())
700}

702/// A partition field: a column, `year|month|day|hour(column)` (with Iceberg's aliases), or
703/// `bucket|truncate(N, column)`.
704fn partition_field(expr: &Expr) -> PolarsResult<CatalogPartitionField> {
705    let unsupported = || {
706        polars_err!(
707            SQLInterface: "unsupported partition field `{}`; expected a column, or year, month, day, hour, bucket(N, column) or truncate(W, column) of a column",
708            expr
709        )
710    };
711    let column = |expr: &Expr| match unparenthesized(expr) {
712        Expr::Identifier(ident) => Some(ident.value.clone()),
713        _ => None,
714    };
715    // A qualified column or a struct field: without the schema, these look the same.
716    let is_compound = |expr: &Expr| matches!(unparenthesized(expr), Expr::CompoundIdentifier(_));
717    let compound = |expr: &Expr| {
718        polars_err!(
719            SQLInterface: "`PARTITION BY` takes unqualified column names; found `{}`; partitioning by a nested field is not yet supported",
720            unparenthesized(expr)
721        )
722    };
723    if let Some(column) = column(expr) {
724        return Ok(CatalogPartitionField {
725            column,
726            transform: "identity".to_string(),
727        });
728    }
729    if is_compound(expr) {
730        return Err(compound(expr));
731    }
732    let Expr::Function(Function {
733        name,
734        uses_odbc_syntax: false,
735        parameters: FunctionArguments::None,
736        args: FunctionArguments::List(list),
737        filter: None,
738        null_treatment: None,
739        over: None,
740        within_group,
741    }) = unparenthesized(expr)
742    else {
743        return Err(unsupported());
744    };
745    let fname = match name_parts(name).as_deref() {
746        Some([fname])
747            if within_group.is_empty()
748                && list.duplicate_treatment.is_none()
749                && list.clauses.is_empty() =>
750        {
751            fname.to_ascii_lowercase()
752        },
753        _ => return Err(unsupported()),
754    };
755    let args = list
756        .args
757        .iter()
758        .map(|arg| match arg {
759            FunctionArg::Unnamed(FunctionArgExpr::Expr(expr)) => Some(expr),
760            _ => None,
761        })
762        .collect::<Option<Vec<_>>>()
763        .ok_or_else(unsupported)?;
764    let positive_int = |expr: &Expr| match unparenthesized(expr) {
765        Expr::Value(ValueWithSpan {
766            value: Value::Number(n, false),
767            ..
768        }) => n.parse::<i32>().ok().filter(|n| *n > 0),
769        _ => None,
770    };
771    let (transform, source) = match (fname.as_str(), args.as_slice()) {
772        ("year" | "years", [source]) => ("year".to_string(), source),
773        ("month" | "months", [source]) => ("month".to_string(), source),
774        ("day" | "days", [source]) => ("day".to_string(), source),
775        ("hour" | "hours", [source]) => ("hour".to_string(), source),
776        ("bucket" | "truncate", [n, source]) => {
777            let Some(n) = positive_int(n) else {
778                if (column(n).is_some() || is_compound(n)) && positive_int(source).is_some() {
779                    polars_bail!(SQLInterface: "`{}` takes the number first, as in `{}({}, {})`", expr, fname, source, n)
780                }
781                polars_bail!(SQLInterface: "the first argument of `{}` must be an integer from 1 to {}", expr, i32::MAX)
782            };
783            (format!("{fname}[{n}]"), source)
784        },
785        _ => return Err(unsupported()),
786    };
787    let column = match column(source) {
788        Some(column) => column,
789        None if is_compound(source) => return Err(compound(source)),
790        None => return Err(unsupported()),
791    };
792    Ok(CatalogPartitionField { column, transform })
793}
794
795/// `expr` without enclosing parentheses.
796fn unparenthesized(mut expr: &Expr) -> &Expr {
797    while let Expr::Nested(inner) = expr {
798        expr = inner;
799    }
800    expr
801}

803/// The properties of a catalog namespace, from `WITH (...)`.
804pub(crate) fn create_schema_options(options: &[SqlOption]) -> PolarsResult<Vec<(String, String)>> {
805    let mut properties: Vec<(String, String)> = Vec::new();
806    for option in options {
807        let (key, value) = option_key_value(option, "schema", &["location"])?;
808        polars_ensure!(
809            key != "location" || is_absolute_location(&value),
810            SQLInterface: "location must be an absolute path or a URI; found '{}'", value
811        );
812        polars_ensure!(
813            properties.iter().all(|(k, _)| k != key),
814            SQLInterface: "duplicate schema option '{}'", key
815        );
816        properties.push((key.to_string(), value));
817    }
818    Ok(properties)
819}

821/// The key and value of a `'key' = value` option of a `kind` ("table", "schema"), with keys that
822/// differ from a `known` one only in case rejected.
823fn option_key_value<'a>(
824    option: &'a SqlOption,
825    kind: &str,
826    known: &[&str],
827) -> PolarsResult<(&'a str, String)> {
828    let SqlOption::KeyValue { key, value } = option else {
829        polars_bail!(SQLInterface: "{} options must be `'key' = value` pairs; found `{}`", kind, option)
830    };
831    let key = key.value.as_str();
832    polars_ensure!(!key.is_empty(), SQLInterface: "{} option names must not be empty", kind);
833    for known in known {
834        polars_ensure!(
835            key == *known || !key.eq_ignore_ascii_case(known),
836            SQLInterface: "'{}' is not a {} option; did you mean '{}'?", key, kind, known
837        );
838    }
839    let Some(value) = option_value(value) else {
840        polars_bail!(
841            SQLInterface: "{} option values must be string, number or boolean literals; found `{}` for '{}'",
842            kind, value, key
843        )
844    };
845    Ok((key, value))
846}
847
848/// A table option value as a property string, if it is a string, number or boolean literal.
849fn option_value(value: &Expr) -> Option<String> {
850    match value {
851        Expr::Value(v) => match &v.value {
852            Value::SingleQuotedString(s) => Some(s.clone()),
853            Value::Number(n, _) => Some(n.clone()),
854            Value::Boolean(b) => Some(b.to_string()),
855            _ => None,
856        },
857        Expr::UnaryOp {
858            op: UnaryOperator::Minus,
859            expr,
860        } => match expr.as_ref() {
861            Expr::Value(ValueWithSpan {
862                value: Value::Number(n, _),
863                ..
864            }) => Some(format!("-{n}")),
865            _ => None,
866        },
867        _ => None,
868    }
869}

871/// Whether a location is an absolute path (POSIX or Windows) or has a URI scheme.
872fn is_absolute_location(location: &str) -> bool {
873    let bytes = location.as_bytes();
874    let windows_path = bytes.len() > 2
875        && bytes[0].is_ascii_alphabetic()
876        && bytes[1] == b':'
877        && matches!(bytes[2], b'/' | b'\\');
878    let has_scheme = location.split_once("://").is_some_and(|(scheme, _)| {
879        scheme.starts_with(|c: char| c.is_ascii_alphabetic())
880            && scheme
881                .chars()
882                .all(|c| c.is_ascii_alphanumeric() || matches!(c, '+' | '-' | '.'))
883    });
884    location.starts_with('/') || windows_path || has_scheme
885}

887/// A table name for messages, with parts containing dots quoted.
888pub(crate) fn display_name(parts: &[String]) -> String {
889    let parts: Vec<String> = parts
890        .iter()
891        .map(|part| match part.contains('.') {
892            true => format!("\"{part}\""),
893            false => part.clone(),
894        })
895        .collect();
896    parts.join(".")
897}
